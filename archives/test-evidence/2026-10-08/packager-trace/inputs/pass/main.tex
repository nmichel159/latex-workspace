% Venue-neutral article wrapper: the default template for new articles.
% This is the ONLY template-dependent file. To move the manuscript to a journal template, replace this
% file and add the journal's .cls/.bst. sections/, references.bib and img/ stay unchanged; preamble/ loses
% the lines the class already provides (amsthm, hyperref, cleveref, natbib, theorem environments).
% House rules: knowledge/writing/latex-conventions.md   Porting: knowledge/writing/template-porting.md
% Build: .\scripts\build-project.ps1 -Project <project>

% No automatic EPS-to-PDF conversion (figures are PDF/PNG). Also avoids a failure in this MiKTeX:
% cleveref with options makes epstopdf-base require grfext.sty, which is not installed.
\newcommand*{\DoNotLoadEpstopdf}{}
\documentclass[11pt,a4paper]{article}

% --- Fonts and page layout (template-dependent) ------------------------------------------------
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{microtype}
\usepackage[margin=2.5cm]{geometry}
% Latin Modern has no bold small caps (T1/lmr/bx/sc). Use medium small caps in bold contexts
% (problem-box titles, headings) instead of LaTeX's bold-upright fallback and its font warning.
\AtBeginDocument{\DeclareFontShape{T1}{lmr}{bx}{sc}{<->ssub*lmr/m/sc}{}}

% --- Title and author: defined once, used by \title, \author and the PDF metadata --------------
\newcommand{\PaperTitle}{Title of the Paper}

\hypersetup{pdftitle={\PaperTitle}, pdfauthor={\PaperAuthor}}
\usepackage[capitalise,noabbrev]{cleveref}   % after hyperref, before any \newtheorem
\input{preamble/environments}
\input{preamble/macros}
